# Backend (Django + DRF) dependencies. There was no manifest for these
# yet -- Phase 1-3 apparently installed them straight into
# .venv-django without recording versions anywhere. Pinned here to
# what's actually in that venv, plus the two new Phase 4 packages.
#
#   pip install -r backend/requirements.txt
Django==6.1.1
djangorestframework==3.18.0

# Phase 4 -- S3-compatible object storage for the Document model
# (docs/DJANGO_MIGRATION_PLAN.md Phase 4). django-storages provides
# the Storage backend; boto3 is its S3 client dependency.
django-storages[s3]==1.14.4
boto3==1.35.36

# Phase 6 -- test-only. postings/tests/test_extraction.py reads the
# real fixture PDFs the same way _app/requirements.txt's pypdf pin
# always has (see that file); not needed at runtime.
pypdf>=4.0

# Phase 9 -- Gmail OAuth (email_sync/oauth.py, email_sync/gmail_provider.py).
# google-api-python-client is the Gmail v1 client gmail_provider.py's
# service_factory-supplied object duck-types (see that module's own
# docstring on why it never imports this itself); google-auth +
# google-auth-oauthlib are the token/credential and browser-consent-
# flow libraries oauth.py builds on directly. cryptography backs
# oauth.py's field-level encryption of stored OAuth tokens
# (GmailCredential.access_token/refresh_token). requests is pinned
# explicitly (not just relied on as a transitive dep of the google-*
# packages above) because oauth.py's disconnect/revoke flow calls it
# directly, against Google's OAuth revoke endpoint.
google-api-python-client==2.200.0
google-auth==2.57.1
google-auth-oauthlib==1.4.1
cryptography>=42.0
requests>=2.32

# Phase 9 (second provider) -- Outlook/Microsoft Graph
# (email_sync/outlook_oauth.py, email_sync/outlook_provider.py). No
# new package beyond what's already pinned above: the Microsoft
# identity platform's v2.0 authorization-code flow is three plain
# REST calls (see outlook_oauth.py's own module docstring for why
# that's `requests` directly rather than the `msal` SDK), and Graph
# itself is a plain REST API `outlook_provider.py` calls the same way.

# Phase 9 (third provider) -- generic IMAP (email_sync/imap_auth.py,
# email_sync/imap_provider.py). No new package at all: `imaplib` (RFC
# 3501 client) is part of the Python standard library, and
# `cryptography` (already pinned above for Gmail/Outlook's own
# field-level token encryption) covers IMAPCredential.password's
# encryption too.

# Phase 10 -- background task scheduling (config/celery.py,
# email_sync/tasks.py), closing the "no background task runner" gap
# docs/DJANGO_MIGRATION_PLAN.md's Phase 9 Known gaps and
# docs/DJANGO_BACKEND_HANDOFF.md §4 both flag. Celery workers/beat run
# as separate processes alongside manage.py runserver (see
# docs/DJANGO_MIGRATION_PLAN.md Phase 10's "background worker process"
# bullet); Redis is both the broker and, for now, the result backend
# -- one fewer moving part than a second Postgres-backed store, and
# nothing here queries a task's result after the fact yet. Unlike
# every version above, pinned exactly against a real installed venv
# this project already had, these two are minimum-version bounds, not
# exact pins: this sandbox has no network access to install or freeze
# a real version against this session (same constraint every earlier
# checkpoint on this track has flagged), so there's nothing real to
# snapshot here yet. Run `pip install -r requirements.txt` normally,
# then `pip freeze | grep -iE 'celery|redis'` and replace these two
# lines with exact pins the same way Django/DRF/the OAuth packages
# above already are.
celery>=5.4
redis>=5.0
